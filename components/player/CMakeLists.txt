cmake_minimum_required(VERSION 3.24)
project(ff_player_native VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FF_WITH_CEF "Build the Linux CEF presentation runtime" OFF)
find_package(SQLite3 REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
  add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
find_package(Threads REQUIRED)
add_library(ff_core src/json.cpp src/store.cpp src/playback.cpp src/playlist_update.cpp src/protocol.cpp src/server.cpp src/media_kind.cpp src/input.cpp src/overlay.cpp src/display.cpp)
target_include_directories(ff_core PUBLIC include vendor)
target_link_libraries(ff_core PUBLIC SQLite3::SQLite3 Threads::Threads)
target_compile_options(ff_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(ff-player-core src/core_main.cpp)
target_link_libraries(ff-player-core PRIVATE ff_core)
add_executable(ff-player-import src/import_main.cpp)
target_link_libraries(ff-player-import PRIVATE ff_core)
install(TARGETS ff-player-import RUNTIME DESTINATION lib/ff-player-native)
enable_testing()
add_executable(frame-clock-tests tests/frame_clock_test.cpp)
target_include_directories(frame-clock-tests PRIVATE include)
add_test(NAME frame-clock COMMAND frame-clock-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME migration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/migration_test.py $<TARGET_FILE:ff-player-import>)
add_executable(core-tests tests/core_test.cpp)
target_link_libraries(core-tests PRIVATE ff_core)
add_test(NAME core COMMAND core-tests ${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures.json)
add_executable(server-tests tests/server_test.cpp)
target_link_libraries(server-tests PRIVATE ff_core)
add_test(NAME server COMMAND server-tests ${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures.json)
add_executable(input-tests tests/input_test.cpp)
target_link_libraries(input-tests PRIVATE ff_core)
add_test(NAME input COMMAND input-tests)
add_executable(scheduling-tests tests/scheduling_test.cpp)
target_link_libraries(scheduling-tests PRIVATE ff_core)
add_test(NAME scheduling COMMAND scheduling-tests)
add_executable(recovery-tests tests/recovery_test.cpp)
target_link_libraries(recovery-tests PRIVATE ff_core)
add_test(NAME recovery COMMAND recovery-tests)
add_executable(modes-tests tests/modes_test.cpp)
target_link_libraries(modes-tests PRIVATE ff_core)
add_test(NAME modes COMMAND modes-tests)
add_executable(playlist-update-tests tests/playlist_update_test.cpp)
target_link_libraries(playlist-update-tests PRIVATE ff_core)
add_test(NAME playlist-update COMMAND playlist-update-tests)
add_executable(overlay-tests tests/overlay_test.cpp)
target_link_libraries(overlay-tests PRIVATE ff_core)
add_test(NAME overlay COMMAND overlay-tests)
add_executable(display-tests tests/display_test.cpp)
target_link_libraries(display-tests PRIVATE ff_core)
add_test(NAME display COMMAND display-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(ff_media_ipc src/media_ipc.cpp)
  target_link_libraries(ff_media_ipc PUBLIC ff_core)
  add_executable(media-ipc-tests tests/media_ipc_test.cpp)
  target_link_libraries(media-ipc-tests PRIVATE ff_media_ipc)
  add_test(NAME media-ipc COMMAND media-ipc-tests)
endif()
if(FF_WITH_CEF)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The presentation runtime targets FFOS Linux")
  endif()
  include(cmake/cef.cmake)
endif()
